import MarketKit
import Testing
@testable import Unstoppable

struct CreateAccountViewModelTests {
    @Test func defaultTokensIncludeUsdcOnEthereumMainnet() {
        let usdcEthereum = "0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48"

        let hasUsdc = CreateAccountViewModel.defaultTokenQueries.contains { query in
            guard query.blockchainType == .ethereum else { return false }
            guard case let .eip20(address) = query.tokenType else { return false }
            return address.caseInsensitiveCompare(usdcEthereum) == .orderedSame
        }

        #expect(hasUsdc)
    }

    @Test func defaultTokensStoreUsdcAddressLowercased() {
        let usdcEthereumLower = "0xa0b86991c6218b36c1d19d4a2e9eb0ce3606eb48"

        let hasLowercased = CreateAccountViewModel.defaultTokenQueries.contains { query in
            guard query.blockchainType == .ethereum else { return false }
            guard case let .eip20(address) = query.tokenType else { return false }
            return address == usdcEthereumLower
        }

        #expect(hasLowercased)
    }
}
